ThemeContext.jsx - Sistema Tema (Fragmento)\n\n**Archivo:** `frontend/src/context/ThemeContext.jsx`  \n**Propósito:** Contexto React para tema claro/oscuro  \n**Líneas totales:** 80+\n\n```jsx\nimport React, { createContext, useContext, useState, useEffect } from 'react';\nimport { Sun, Moon } from 'lucide-react';\n\nconst ThemeContext = createContext();\n\nexport function ThemeProvider({ children }) {\n  const [theme, setTheme] = useState('light');\n\n  useEffect(() => {\n    // Leer localStorage o detectar preferencia del sistema\n    const savedTheme = localStorage.getItem('theme');\n    if (savedTheme) {\n      setTheme(savedTheme);\n      document.documentElement.className = savedTheme;\n    } else if (window.matchMedia('(prefers-color-scheme: dark)').matches) {\n      setTheme('dark');\n      document.documentElement.className = 'dark';\n    } else {\n      setTheme('light');\n      document.documentElement.className = 'light';\n    }\n  }, []);\n\n  const toggleTheme = () => {\n    const newTheme = theme === 'light' ? 'dark' : 'light';\n    setTheme(newTheme);\n    document.documentElement.className = newTheme;\n    localStorage.setItem('theme', newTheme);\n  };\n\n  return (\n    <ThemeContext.Provider value={{ theme, toggleTheme }}>\n      {children}\n    </ThemeContext.Provider>\n  );\n}\n\nexport function useTheme() {\n  const context = useContext(ThemeContext);\n  if (!context) {\n    throw new Error('useTheme debe ser usado dentro de ThemeProvider');\n  }\n  return context;\n}\n\nexport function ThemeToggle() {\n  const { theme, toggleTheme } = useTheme();\n  return (\n    <button\n      onClick={toggleTheme}\n      className=\"fixed top-4 right-4 z-50 p-2 rounded-lg glass-panel-sm hover:bg-slate-200 dark:hover:bg-slate-800 transition\"\n    >\n      {theme === 'light' ? <Moon className=\"h-5 w-5\" /> : <Sun className=\"h-5 w-5\" />}\n    </button>\n  );\n}\n```\n\n**Características:**\n- ✅ `createContext` para estado global\n- ✅ localStorage persistence\n- ✅ Detección `prefers-color-scheme`\n- ✅ Aplica clase a `document.documentElement`\n- ✅ Hook `useTheme` para acceso en componentes\n- ✅ Componente `ThemeToggle` para botón\n\n---\n\n## 5.
